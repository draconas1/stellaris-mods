sploit_mining_terminal = {
	icon = "GFX_starbase_mining_terminal"
	section = "REFINERY_STARBASE_SECTION"
	construction_days = 180
	category = sm_economy_cat
	initial = yes

	potential = {
		OR = {
			is_normal_starbase = yes
			is_orbital_ring = yes
		}
	}

	possible = {
		custom_tooltip = {
			fail_text = "requires_mineral_station"
			solar_system = {
				any_system_planet = {
					has_mining_station = yes
					OR = {
						has_resource = { type = minerals amount > 0 }
						has_resource = { type = alloys amount > 0 }
					}
				}
			}
		}
	}

	resources = {
		category = starbase_modules
		cost = {
			alloys = 50
		}
		upkeep = {
			energy = 1
		}
		produces = {
			minerals = 6
		}
	}

	show_tech_unlock_if = {
		is_nomadic = no
	}

	ai_build_at_chokepoint = no
	ai_build_outside_chokepoint = yes
	ai_weight = {
		weight = 100
	}
}

	initial = yes

	potential = {
		exists = owner
		is_orbital_ring = yes

			has_technology = "tech_solar_panel_network"

			alloys = 50
		}

		produces = {
			energy = 6
		}

		produces = {
			trigger = {

				}
