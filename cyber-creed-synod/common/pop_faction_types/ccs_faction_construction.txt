ccs_cyber_creed_construction_xenophobe = {
    election_header = "GFX_faction_header_yellow"
    guiding_ethic = ethic_xenophobe
    visual_ethic = "ccs_construction_xenophobe_visual"
    use_guiding_ethic_as_pop_filter = no
    use_guiding_ethic_as_growth_factor = no

    persistent_faction = yes
    can_rename = no
    unique = yes
    support_multiplier = 0.33

    skip_creation_cooldown = yes

    resources = {
        category = pop_factions
        inline_script = pop_faction_types/shared_unity_production
        produces = {
            consumer_goods = @faction_major_output
        }
        produces = {
            trigger = {
                owner = {
                    OR = {
                        has_valid_civic = civic_crowdsourcing
                        has_valid_civic = civic_corporate_crowdsourcing
                    }
                }
            }
            engineering_research = @faction_major_output
            society_research = @faction_minor_output
            mult = @crowdsourcing_multiplier
        }
    }

    extortion_resource = {
        category = pop_factions
        produces = {
            engineering_research = @faction_major_output
            society_research = @faction_minor_output
            mult = @faction_extortion_crowdsourcing_ratio
        }
    }

    is_potential = {
        has_origin = origin_cybernetic_creed
        has_country_flag = cybernetic_creed_ccs
        has_country_flag = ccs_cyber_creed_construction_xenophobe
    }

    can_join_pre_triggers = {
        is_enslaved = no
        is_being_purged = no
    }

    can_join_faction = {
        inline_script = pop_faction_types/can_join_faction/base_shackled
    }

    inline_script = pop_faction_types/attraction/promote_suppress_tags

    attraction = {
        base = 100

        inline_script = pop_faction_types/attraction/promote_suppress_attraction
        inline_script = pop_faction_types/attraction/leader_attraction
    }

    leader = {
        base = 100
        modifier = {
            factor = 0
            species = {
                has_trait = "trait_mechanical"
            }
            owner = {
                is_individual_machine = no
            }
        }

        modifier = {
            factor = 0.0
            is_event_leader = yes
        }
    }


    actions = {
        embrace_faction = {
            title = "EMBRACE_FACTION"
            description = "EMBRACE_FACTION_DESC"

            cost = {
                unity = 5000
            }

            potential = {
                exists = owner
                owner = {
                    OR = {
                        is_subject = no
                        NOT = { any_agreement = { agreement_preset = preset_dominion } }
                    }
                }
            }

            valid = {
                custom_tooltip = {
                    fail_text = EMBRACE_FACTION_COOLDOWN
                    owner = {
                        NOT = { has_modifier = embraced_faction_timer }
                    }
                }
                support > 0.20
                owner = {
                    NOT = { has_ethic = "ethic_fanatic_spiritualist" }
                }
            }

            effect = {
                add_modifier = { modifier = embraced_faction days = 3600 }
                owner = {
                    set_country_flag = has_embraced_faction
                    shift_ethic = ethic_spiritualist
                    hidden_effect = {
                        add_modifier = { modifier = embraced_faction_timer days = 3600 }
                        every_pop_faction = {
                            limit = { NOT = { is_same_value = root } }
                            add_modifier = { modifier = embraced_another_faction days = 3600 }
                        }
                    }
                }
                hidden_effect = {
                    save_event_target_as = TargetFaction
                    owner = {
                        every_relation = {
                            limit = {
                                is_ai = no
                                is_country_type = default
                                has_communications = prev
                                has_intel_level = {
                                    who = prev
                                    category = government
                                    level >= 1
                                }
                            }
                            country_event = { id = factions.2000 }
                        }
                    }
                }
            }

            ai_weight = {
                base = 1
                modifier = {
                    factor = 0
                    support < 0.50
                    owner = {
                        has_ethic = ethic_spiritualist
                    }
                }
                modifier = {
                    factor = 0
                    owner = { has_valid_civic = civic_fanatic_purifiers }
                }
                modifier = {
                    factor = 0
                    owner = {
                        count_pop_faction = {
                            count < 4
                        }
                    }
                }
            }
        }
    }
    inline_script = {
        script = pop_faction_types/actions/promote_faction
        ETHIC = spiritualist
    }
    inline_script = {
        script = pop_faction_types/actions/suppress_faction
        ETHIC = spiritualist
    }
    actions = {
        # Faction Extortion Buttons
        extort_faction = {
            title = "EXTORT_FACTION"
            description = "EXTORT_FACTION_DESC"

            potential = {
                exists = owner
                owner = {
                    is_oligarchic_authority = yes
                    has_country_flag = cyber_collectivist
                }
                is_faction_extorted = no
            }

            effect = {
                set_faction_extorted = yes
                custom_tooltip = engineering_research_extortion_tt
                custom_tooltip = ONLY_ONE_EXTORTED_FACTION_TT
            }

            ai_weight = {
                base = 0
            }
        }
        cancel_extort_faction = {
            title = "CANCEL_EXTORT_FACTION"
            description = "CANCEL_EXTORT_FACTION_DESC"

            potential = {
                exists = owner
                owner = {
                    is_oligarchic_authority = yes
                    has_country_flag = cyber_collectivist
                }
                is_faction_extorted = yes
            }

            effect = {
                set_faction_extorted = no
            }

            ai_weight = {
                base = 0
            }
        }
    }
}
