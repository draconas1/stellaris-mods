country_event = {
    id = ccs_events.100
    title = "ccs_events.100.name"
    desc = "ccs_events.100.desc"
    picture = GFX_evt_cybernetic_creed
    event_chain = cybernetic_creed_chain
    is_triggered_only = yes

    option = {
        allow = { is_egalitarian = no }
        name = "ccs_events.100.authoritarian"
        custom_tooltip = "ccs_events.100.authoritarian.tooltip"
        inline_script = {
            script = ccs/faction_choice
            ETHIC = authoritarian
            FACTION = ccs_cyber_creed_labor_authoritarian
        }
    }

    option = {
        allow = { is_authoritarian = no }
        name = "ccs_events.100.egalitarian"
        custom_tooltip = "ccs_events.100.egalitarian.tooltip"
        inline_script = {
            script = ccs/faction_choice
            ETHIC = egalitarian
            FACTION = ccs_cyber_creed_labor_egalitarian
        }
    }

    after = {
        hidden_effect = {
            country_event = { id = ccs_events.200 days = 1 random = 1  }
        }
    }
}

#choir choice
country_event = {
    id = ccs_events.400
    title = "ccs_events.400.name"
    desc = "ccs_events.400.desc"
    picture = GFX_evt_cybernetic_creed
    event_chain = cybernetic_creed_chain
    is_triggered_only = yes

    option = {
        name = "ccs_events.400.robots"
        custom_tooltip = "ccs_events.400.robots.tooltip"
        set_policy = {
            policy = robot_pop_policy
            option = robot_pops_allowed
            cooldown = yes
        }
        hidden_effect = {
            set_country_flag = ccs_cyber_creed_research_robots
            enable_faction_of_type = ccs_cyber_creed_research_robots
            every_owned_pop_group = { force_faction_evaluation = yes }
        }
    }

    option = {
        name = "ccs_events.400.norobots"
        custom_tooltip = "ccs_events.400.norobots.tooltip"
        set_policy = {
            policy = robot_pop_policy
            option = robot_pops_outlawed
            cooldown = yes
        }
        hidden_effect = {
            set_country_flag = ccs_cyber_creed_research_norobots
            enable_faction_of_type = ccs_cyber_creed_research_norobots
            every_owned_pop_group = { force_faction_evaluation = yes }
        }
    }

    after = {
        hidden_effect = {
            country_event = { id = ccs_events.500 days = 1 }
        }
    }
}
