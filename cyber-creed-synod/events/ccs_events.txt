country_event = {
    id = ccs_events.100
    title = "ccs_events.100.name"
    desc = "ccs_events.100.desc"
    picture = GFX_evt_cybernetic_creed
    event_chain = cybernetic_creed_chain
    is_triggered_only = yes

    option = {
        allow = { is_egalitarian = no }
        name = "ccs_events.100.authoritarian"
        custom_tooltip = "ccs_events.100.authoritarian.tooltip"
        hidden_effect = {






            every_owned_pop_group = {
                limit = { member_of_faction = cyber_creed_labor }
                pop_change_ethic = { ethic = ethic_authoritarian percentage = 1  }
                set_timed_pop_group_flag = { flag = css_ethic_authoritarian_attr days = 1000 }
            }
            set_country_flag = cybernetic_creed_ccs_labor_authoritarian
            enable_faction_of_type = ccs_cyber_creed_labor_2_authortarian
        }
    }

    option = {
        allow = { is_authoritarian = no }
        name = "ccs_events.100.egalitarian"
        custom_tooltip = "ccs_events.100.egalitarian.tooltip"
        every_owned_pop_group = {
            limit = { has_trait = trait_cyborg_ritualistic_implants	}
            random_list = {
                1 = {
                    modify_species = {
                        species = this
                        # remove_trait = trait_cyborg_ritualistic_implants
                        add_trait = trait_cyborg_creed_of_research
                        add_traits_at_start_of_list = yes
                    }
                }
                1 = {
                    modify_species = {
                        species = this
                        #remove_trait = trait_cyborg_ritualistic_implants
                        add_trait = trait_cyborg_creed_of_war
                        add_traits_at_start_of_list = yes
                    }
                }
                1 = {
                    modify_species = {
                        species = this
                        # remove_trait = trait_cyborg_ritualistic_implants
                        add_trait = trait_cyborg_creed_of_construction
                        add_traits_at_start_of_list = yes
                    }
                }
                1 = {
                    modify_species = {
                        species = this
                        # remove_trait = trait_cyborg_ritualistic_implants
                        add_trait = trait_cyborg_creed_of_labor
                        add_traits_at_start_of_list = yes
                    }
                }
            }



        }
        every_owned_pop_group = {
            limit = { member_of_faction = cyber_creed_labor }
            pop_change_ethic = { ethic = ethic_egalitarian percentage = 1  }
            set_timed_pop_group_flag = { flag = css_ethic_egalitarian_attr days = 1000 }
        }
        set_country_flag = cybernetic_creed_ccs_labor_egalitarian
        enable_faction_of_type = ccs_cyber_creed_labor_2_egalitarian
        add_modifier = { modifier = promoted_egalitarian }

        hidden_effect = {

        }
    }

    after = {
        hidden_effect = {
            country_event = { id = ccs_events.200 days = 1 random = 1  }
        }
    }
}
