#choir choice
country_event = {
    id = ccs_events.400
    title = "ccs_events.400.name"
    desc = "ccs_events.400.desc"
    picture = GFX_evt_cybernetic_creed
    event_chain = cybernetic_creed_chain
    is_triggered_only = yes

    option = {
        name = "ccs_events.400.robots"
        custom_tooltip = "ccs_events.400.robots.tooltip"
        set_policy = {
            policy = robot_pop_policy
            option = robot_pops_allowed
            cooldown = yes
        }
        hidden_effect = {
            set_country_flag = cybernetic_creed_ccs_choir_robots
            enable_faction_of_type = ccs_cyber_creed_research
        }
    }

    option = {
        name = "ccs_events.400.norobots"
        custom_tooltip = "ccs_events.400.norobots.tooltip"
        set_policy = {
            policy = robot_pop_policy
            option = robot_pops_outlawed
            cooldown = yes
        }
        hidden_effect = {
            set_country_flag = cybernetic_creed_ccs_choir_norobots
            enable_faction_of_type = ccs_cyber_creed_research
        }
    }

    after = {
        every_owned_pop_group = { force_faction_evaluation = yes }
        hidden_effect = {
            country_event = { id = ccs_events.500 days = 1 }
        }
    }
}

country_event = {
    id = ccs_events.1000
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        inline_script = { script = ccs/count_faction_planets MOD = ccs_commune_planet }
        inline_script = { script = ccs/count_faction_planets MOD = ccs_fellowship_planet }
        inline_script = { script = ccs/count_faction_planets MOD = ccs_choir_planet }
        inline_script = { script = ccs/count_faction_planets MOD = ccs_templar_planet }
        if = {
            limit = { count_owned_colony = { count > 4 } }
            set_country_flag = ccs_faction_squabbles
        }
        set_variable = { which = ccs_min_planets value = value:ccs_lowest_planet_count }
        set_variable = { which = ccs_max_planets value = value:ccs_highest_planet_count }

        inline_script = {
            script = ccs/faction_compare_counts
            FACTION = ccs_commune
            OTHER_1 = ccs_fellowship
            OTHER_2 = ccs_choir
            OTHER_3 = ccs_templar
        }
        inline_script = {
            script = ccs/faction_compare_counts
            FACTION = ccs_fellowship
            OTHER_1 = ccs_commune
            OTHER_2 = ccs_choir
            OTHER_3 = ccs_templar
        }
        inline_script = {
            script = ccs/faction_compare_counts
            FACTION = ccs_choir
            OTHER_1 = ccs_fellowship
            OTHER_2 = ccs_commune
            OTHER_3 = ccs_templar
        }
        inline_script = {
            script = ccs/faction_compare_counts
            FACTION = ccs_templar
            OTHER_1 = ccs_fellowship
            OTHER_2 = ccs_choir
            OTHER_3 = ccs_commune
        }
    }
}
